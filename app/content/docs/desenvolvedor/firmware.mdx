O firmware é desenvolvido em **C/C++ com o framework Arduino** e gerenciado pelo **PlatformIO**. Ele roda no **NodeMCU v3 (ESP8266MOD)** instalado em cada porta e é responsável por conectar ao Wi-Fi, comunicar-se com o broker MQTT e controlar todos os periféricos físicos.

#define WIFI_SSID       "NomeDoWifi"

## Loop principal (`main.cpp`)

O firmware utiliza **temporização não-bloqueante** com `millis()`. O uso de `delay()` é proibido no loop principal — ele bloquearia a leitura de mensagens MQTT e a detecção de eventos de hardware.

```cpp
// core/src/main.cpp
#include <Arduino.h>
#include "config.h"
#include "wifi/wifi.h"
#include "mqtt/mqtt.h"
#include "fingerprint/fingerprint.h"
#include "nfc/nfc.h"
#include "door/door.h"
#include "led/led.h"

// ── Estado global ──────────────────────────────────────────────
enum DeviceMode { MODE_DOOR, MODE_ENROLLMENT };
DeviceMode currentMode = MODE_DOOR;

// ── Temporização ───────────────────────────────────────────────
unsigned long lastHeartbeat   = 0;
unsigned long lastDoorStatus  = 0;
unsigned long relayOpenSince  = 0;
bool          relayOpen       = false;

void setup() {
  Serial.begin(115200);

  pinMode(PIN_RELAY,       OUTPUT);
  pinMode(PIN_REED_SWITCH, INPUT_PULLUP);
  pinMode(PIN_LED_STATUS,  OUTPUT);

  digitalWrite(PIN_RELAY, LOW);  // Fechadura travada por padrão

  wifiConnect(WIFI_SSID, WIFI_PASSWORD, WIFI_TIMEOUT_MS);
  mqttSetup(MQTT_BROKER, MQTT_PORT, MQTT_CLIENT_ID, onMqttMessage);
  fingerprintSetup(PIN_FP_TX, PIN_FP_RX);
  nfcSetup(PIN_NFC_D0, PIN_NFC_D1);

  mqttPublishRegister();  // door/{id}/register
}

void loop() {
  unsigned long now = millis();

  // Manter conexões ativas
  if (!wifiConnected())   wifiReconnect();
  if (!mqttConnected())   mqttReconnect();
  mqttLoop();  // Processa mensagens recebidas

  if (currentMode == MODE_DOOR) {
    // Heartbeat periódico
    if (now - lastHeartbeat >= HEARTBEAT_INTERVAL_MS) {
      mqttPublishHeartbeat();
      lastHeartbeat = now;
    }

    // Status da porta periódico
    if (now - lastDoorStatus >= DOOR_STATUS_INTERVAL_MS) {
      bool doorOpen = (digitalRead(PIN_REED_SWITCH) == LOW);
      bool locked   = !relayOpen;
      mqttPublishDoorStatus(doorOpen ? "OPEN" : "CLOSED", locked);
      lastDoorStatus = now;
    }

    // Fechar relé automaticamente após RELAY_UNLOCK_DURATION_MS
    if (relayOpen && (now - relayOpenSince >= RELAY_UNLOCK_DURATION_MS)) {
      digitalWrite(PIN_RELAY, LOW);
      relayOpen = false;
      ledOff();
    }

    // Leitura de credencial biométrica (evento)
    String fpTemplate = fingerprintRead();
    if (fpTemplate.length() > 0) {
      handleAccessAttempt("FINGERPRINT", fpTemplate);
    }

    // Leitura de credencial NFC (evento via interrupção Wiegand)
    String nfcUid = nfcGetUid();
    if (nfcUid.length() > 0) {
      handleAccessAttempt("NFC_TAG", nfcUid);
    }
  }

  // Modo terminal: lógica de enrollment gerenciada em fingerprint.cpp
  if (currentMode == MODE_ENROLLMENT) {
    enrollmentLoop(now);
  }
}
```

## Modos de operação

### Modo Fechadura (padrão)

Comportamento normal do controlador:

- Processa tentativas de acesso via biometria e NFC
- Envia heartbeat a cada 30 segundos
- Publica o estado da porta a cada 500ms
- Executa comandos recebidos via MQTT (UNLOCK, LOCK, SYNC_STATE)
- Aciona o relé por `RELAY_UNLOCK_DURATION_MS` quando recebe decisão GRANTED

### Modo Terminal (temporário)

Ativado ao receber `door/{id}/enter-enrollment-mode`:

- Para de processar tentativas de acesso (relé permanece inativo)
- LED azul pisca a cada 500ms (aguardando dedo)
- Captura 2 passagens do dedo com o HiLink ZW-101
- Publica `door/{id}/enrollment-result` com o template ou status de falha
- Retorna ao **Modo Fechadura** imediatamente após publicar o resultado
- Expira automaticamente ao atingir `expiresAt`

```cpp
// Alternância de modo via callback MQTT
void onMqttMessage(const char* topic, const byte* payload, unsigned int length) {
  String topicStr = String(topic);
  String body     = String((char*)payload, length);

  if (topicStr == TOPIC_ACCESS_RESULT) {
    handleAccessResult(body);

  } else if (topicStr == TOPIC_COMMAND) {
    handleCommand(body);

  } else if (topicStr == TOPIC_ENTER_ENROLLMENT) {
    currentMode = MODE_ENROLLMENT;
    enrollmentBegin(body);  // Extrai enrollmentId, finger, expiresAt

  } else if (topicStr == TOPIC_CANCEL_ENROLLMENT) {
    enrollmentCancel();
    currentMode = MODE_DOOR;

  } else if (topicStr == TOPIC_SYNC_CREDENTIALS) {
    handleSyncCredentials(body);
  }
}
```

## Fluxo de acesso (Modo Fechadura)

```cpp
void handleAccessAttempt(const char* type, const String& value) {
  // Publica tentativa de acesso
  mqttPublishAccessAttempt(type, value);

  // Aguarda door/{id}/access-result por até ACCESS_RESULT_TIMEOUT_MS
  unsigned long waitStart = millis();
  accessResultReceived    = false;

  while (!accessResultReceived && (millis() - waitStart < ACCESS_RESULT_TIMEOUT_MS)) {
    mqttLoop();
    yield();
  }

  if (!accessResultReceived) {
    // Timeout → política offline (HW-008, pendente)
    ledRed(2000);
    return;
  }

  if (lastAccessDecision == "GRANTED") {
    digitalWrite(PIN_RELAY, HIGH);  // Abre a fechadura
    relayOpen      = true;
    relayOpenSince = millis();
    ledGreen();
    buzzerShortBeep();
  } else {
    ledRed(2000);
    buzzerLongBeep();
  }
}
```

## Fluxo de enrollment (Modo Terminal)

```cpp
// fingerprint/fingerprint.cpp

EnrollmentState enrollState;

void enrollmentBegin(const String& jsonPayload) {
  // Extrai campos do JSON
  enrollState.enrollmentId = extractField(jsonPayload, "enrollmentId");
  enrollState.finger       = extractField(jsonPayload, "finger");
  enrollState.expiresAt    = extractField(jsonPayload, "expiresAt");
  enrollState.step         = ENROLL_STEP_WAITING_FIRST;

  ledBlueBlinkStart(ENROLLMENT_LED_BLINK_MS);
}

void enrollmentLoop(unsigned long now) {
  // Verifica expiração
  if (isExpired(enrollState.expiresAt, now)) {
    publishEnrollmentResult(enrollState.enrollmentId, "", 0, "EXPIRED");
    enrollmentReset();
    currentMode = MODE_DOOR;
    return;
  }

  // Máquina de estados do HiLink ZW-101
  switch (enrollState.step) {
    case ENROLL_STEP_WAITING_FIRST:
      if (fingerprintGenImg() == OK) {
        fingerprintImg2Tz(1);
        enrollState.step = ENROLL_STEP_WAITING_SECOND;
        ledBlueSolid();  // Indica sucesso da primeira passagem
      }
      break;

    case ENROLL_STEP_WAITING_SECOND:
      if (fingerprintGenImg() == OK) {
        fingerprintImg2Tz(2);
        enrollState.step = ENROLL_STEP_CREATE_MODEL;
      }
      break;

    case ENROLL_STEP_CREATE_MODEL: {
      uint8_t result = fingerprintCreateModel();
      if (result != OK) {
        publishEnrollmentResult(enrollState.enrollmentId, "", 0, "FAILED");
        enrollmentReset();
        currentMode = MODE_DOOR;
        return;
      }

      // Extrai template (256 bytes → 512 chars hex)
      uint8_t templateBuffer[256];
      uint8_t quality = 0;
      fingerprintGetModel(templateBuffer, &quality);
      String templateHex = bytesToHex(templateBuffer, 256);

      publishEnrollmentResult(enrollState.enrollmentId, templateHex, quality, "SUCCESS");
      enrollmentReset();
      currentMode = MODE_DOOR;
      break;
    }
  }
}
```

## Temporização não-bloqueante

Regra fundamental: **nunca usar `delay()` no `loop()`**. Use sempre `millis()`:

```cpp
// ✅ CORRETO — não bloqueia o loop
unsigned long lastHeartbeat = 0;
const unsigned long HEARTBEAT_INTERVAL = 30000;

void loop() {
  unsigned long now = millis();
  if (now - lastHeartbeat >= HEARTBEAT_INTERVAL) {
    sendHeartbeat();
    lastHeartbeat = now;
  }
  // Outras tarefas continuam rodando normalmente
}

// ❌ ERRADO — bloqueia o loop inteiro por 30 segundos
void loop() {
  sendHeartbeat();
  delay(30000);  // Nenhuma mensagem MQTT é processada durante este tempo
}
```

## Publicação MQTT

Todas as publicações usam JSON construído manualmente (sem biblioteca de JSON para economizar RAM no ESP8266):

```cpp
void mqttPublishRegister() {
  char payload[256];
  snprintf(payload, sizeof(payload),
    "{\"roomId\":\"%s\",\"sensorProtocol\":\"%s\",\"sensorModel\":\"%s\"}",
    ROOM_ID, SENSOR_PROTOCOL, SENSOR_MODEL
  );
  mqttClient.publish(TOPIC_REGISTER, payload, /*retained=*/false);
}

void mqttPublishAccessAttempt(const char* type, const String& value) {
  char payload[600];
  snprintf(payload, sizeof(payload),
    "{\"credentialType\":\"%s\",\"credentialValue\":\"%s\"}",
    type, value.c_str()
  );
  mqttClient.publish(TOPIC_ACCESS_ATTEMPT, payload);
}

void mqttPublishDoorStatus(const char* doorState, bool isLocked) {
  char payload[64];
  snprintf(payload, sizeof(payload),
    "{\"doorState\":\"%s\",\"isLocked\":%s}",
    doorState, isLocked ? "true" : "false"
  );
  mqttClient.publish(TOPIC_STATUS, payload);
}
```

> ℹ️ Para parsing de JSON recebido (payloads dos comandos), use a biblioteca **ArduinoJson** com `DynamicJsonDocument` dimensionado ao tamanho esperado do payload.

## Reconexão automática

```cpp
void mqttReconnect() {
  int attempts = 0;
  while (!mqttClient.connected() && attempts < 5) {
    if (mqttClient.connect(MQTT_CLIENT_ID)) {
      // Reinscrever em todos os tópicos de entrada
      mqttClient.subscribe(TOPIC_ACCESS_RESULT);
      mqttClient.subscribe(TOPIC_COMMAND);
      mqttClient.subscribe(TOPIC_ENTER_ENROLLMENT);
      mqttClient.subscribe(TOPIC_CANCEL_ENROLLMENT);
      mqttClient.subscribe(TOPIC_SYNC_CREDENTIALS);

      // Reanunciar presença
      mqttPublishRegister();
    } else {
      delay(2000);  // delay permitido aqui — estamos fora do loop normal
      attempts++;
    }
  }
}
```
